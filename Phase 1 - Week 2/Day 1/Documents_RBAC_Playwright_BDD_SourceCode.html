pages/DocumentsPage.ts

import { Page, Locator, expect } from '@playwright/test';

export class DocumentsPage {
  readonly page: Page;
  readonly username: Locator;
  readonly password: Locator;
  readonly loginBtn: Locator;
  readonly roleLabel: Locator;
  readonly adminMenu: Locator;
  readonly fullAccessPanel: Locator;
  readonly accessDeniedMsg: Locator;

  constructor(page: Page) {
    this.page = page;

    // Locators
    this.username = page.locator('#username');
    this.password = page.locator('#password');
    this.loginBtn = page.locator('button[type="submit"]');
    this.roleLabel = page.locator('[data-testid="user-role"]');
    this.adminMenu = page.locator('[data-testid="admin-rules-menu"]');
    this.fullAccessPanel = page.locator('[data-testid="full-access-controls"]');
    this.accessDeniedMsg = page.locator('text=Access Denied');
  }

  async navigate() {
    await this.page.goto('https://documents-app.example.com');
    await expect(this.username).toBeVisible();
  }

  async login(user: string, pwd: string) {
    await this.username.waitFor({ state: 'visible', timeout: 10000 });
    await this.username.fill(user);

    await this.password.waitFor({ state: 'visible', timeout: 10000 });
    await this.password.fill(pwd);

    await this.loginBtn.waitFor({ state: 'visible', timeout: 10000 });
    await this.loginBtn.click();
  }

  async getRole(): Promise {
    await this.roleLabel.waitFor({ state: 'visible', timeout: 10000 });
    return (await this.roleLabel.textContent()) || '';
  }
}

steps/documents.steps.ts

import { Given, When, Then } from '@cucumber/cucumber';
import { expect } from '@playwright/test';
import { DocumentsPage } from '../pages/DocumentsPage';

let documentsPage: DocumentsPage;
let page: any;

Given('the user navigates to the Documents application', async function () {
  try {
    documentsPage = new DocumentsPage(page);
    await documentsPage.navigate();
    await expect(documentsPage.username).toBeVisible();
  } catch (error) {
    console.error('Navigation failed', error);
    throw error;
  }
});

When('the user logs in with {string} credentials', async function (userType) {
  try {
    const users:any = {
      admin: { user: 'adminUser', pwd: 'Password123' },
      nonadmin: { user: 'standardUser', pwd: 'Password123' }
    };

    await documentsPage.login(users[userType].user, users[userType].pwd);
    await expect(documentsPage.roleLabel).toBeVisible();
  } catch (error) {
    console.error('Login failed', error);
    throw error;
  }
});

Then('validate the logged in user role is {string}', async function (role) {
  const actualRole = await documentsPage.getRole();
  expect(actualRole.trim()).toBe(role);
});

Then('validate the logged in user role is not {string}', async function (role) {
  const actualRole = await documentsPage.getRole();
  expect(actualRole.trim()).not.toBe(role);
});

Then('validate admin access page is visible', async function () {
  await documentsPage.adminMenu.waitFor({ state: 'visible', timeout: 10000 });
  await expect(documentsPage.adminMenu).toBeVisible();
});

Then('validate full access controls are available', async function () {
  await documentsPage.fullAccessPanel.waitFor({ state: 'visible', timeout: 10000 });
  await expect(documentsPage.fullAccessPanel).toBeVisible();
});

Then('validate admin access page is not visible', async function () {
  await expect(documentsPage.adminMenu).not.toBeVisible();
});

Then('validate access denied message is displayed', async function () {
  await documentsPage.accessDeniedMsg.waitFor({ state: 'visible', timeout: 10000 });
  await expect(documentsPage.accessDeniedMsg).toBeVisible();
});
